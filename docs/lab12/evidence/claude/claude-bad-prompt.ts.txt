import { goalCategoryEnum, goalStatusEnum } from "../db/schema.js";

export const GOAL_STATUSES = goalStatusEnum.enumValues;
export type GoalStatus = (typeof GOAL_STATUSES)[number];
export function isGoalStatus(value: string): value is GoalStatus {
  return (GOAL_STATUSES as readonly string[]).includes(value);
}
export type GoalUpdateFields = GoalFields | "status";
export type GoalUpdateErrors = Partial<Record<GoalUpdateFields | "form", string>>;
export type GoalUpdate = Partial<GoalInput & { status: GoalStatus }>;

export function validateGoalUpdate(
  input: Record<string, unknown>,
  options: { today?: string; currentDeadline?: string | null } = {},
): { errors: GoalUpdateErrors; changes: GoalUpdate } {
  const today = options.today ?? todayInAppZone();
  const errors: GoalUpdateErrors = {};
  const changes: GoalUpdate = {};

  if (input.title !== undefined) {
    const title = typeof input.title === "string" ? input.title.trim() : "";
    if (title.length === 0) errors.title = "Зорилгын нэр хоосон байж болохгүй";
    else if (title.length > GOAL_TITLE_MAX_LENGTH)
      errors.title = `Нэр ${GOAL_TITLE_MAX_LENGTH} тэмдэгтээс урт байж болохгүй`;
    else changes.title = title;
  }

  if (input.description !== undefined) {
    if (input.description === null) changes.description = null;
    else if (typeof input.description !== "string") errors.description = "Тайлбар буруу байна";
    else {
      const trimmed = input.description.trim();
      if (trimmed.length > GOAL_DESCRIPTION_MAX_LENGTH)
        errors.description = `Тайлбар ${GOAL_DESCRIPTION_MAX_LENGTH} тэмдэгтээс урт байж болохгүй`;
      else changes.description = trimmed.length > 0 ? trimmed : null;
    }
  }

  if (input.deadline !== undefined) {
    const unchanged =
      typeof input.deadline === "string" &&
      options.currentDeadline != null &&
      input.deadline.trim() === options.currentDeadline;
    if (unchanged) changes.deadline = options.currentDeadline;
    else {
      const { error, deadline } = validateDeadline(input.deadline, today);
      if (error !== undefined) errors.deadline = error;
      else changes.deadline = deadline;
    }
  }

  if (input.category !== undefined) {
    if (input.category !== null && typeof input.category !== "string")
      errors.category = "Ангилал буруу байна";
    else {
      const trimmed = (input.category ?? "").trim();
      if (trimmed.length === 0) errors.category = "Ангилал хоосон байж болохгүй";
      else if (!isGoalCategory(trimmed))
        errors.category = `Ангилал нь ${GOAL_CATEGORIES.join(", ")}-ийн аль нэг байх ёстой`;
      else changes.category = trimmed;
    }
  }

  if (input.status !== undefined) {
    if (input.status !== null && typeof input.status !== "string") errors.status = "Төлөв буруу байна";
    else {
      const trimmed = (input.status ?? "").trim();
      if (trimmed.length === 0) errors.status = "Төлөв хоосон байж болохгүй";
      else if (!isGoalStatus(trimmed))
        errors.status = `Төлөв нь ${GOAL_STATUSES.join(", ")}-ийн аль нэг байх ёстой`;
      else changes.status = trimmed;
    }
  }

  if (Object.keys(errors).length === 0 && Object.keys(changes).length === 0)
    errors.form = "Өөрчлөх талбар алга";

  return { errors, changes };
}
